/**
 * Ensure we only output one JSON-LD per @type/@id.
 * If no @id is present, we keep the first occurrence of that @type.
 */
function uniqueByTypeId(nodes: any[]): any[] {
  const seen = new Set<string>();
  return nodes.filter((n) => {
    if (!n || typeof n !== "object") return false;
    const rawType = Array.isArray(n["@type"]) ? n["@type"].join("|") : String(n["@type"] ?? "");
