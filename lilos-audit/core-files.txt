/** Page meta input */
type MetaInput = Partial<MetaDataType> & {
  title?: string;
  description?: string;
};

export interface Props {
  title?: string;
  description?: string;
  ignoreTitleTemplate?: boolean;
  lang?: string;
  bodyClass?: string;
  metadata?: MetaInput;
}

const {
  title,
  description,
  ignoreTitleTemplate,
  lang = "en",
  bodyClass = "",
  metadata = {},
} = Astro.props as Props;

/** Conservative fallback titles when a page doesn't provide one */
const pathname = Astro.url.pathname;
let fallbackTitle: string | undefined;
if (!title) {
  if (pathname === "/" || pathname === "/index") {
    fallbackTitle = "Local SEO & Websites for Home Service Pros";
  } else if (pathname.startsWith("/additional-services")) {
    fallbackTitle = "Additional Services for Home Service Pros";
  }
}
const effectiveTitle = title ?? fallbackTitle;

/** Narrow, safe global surface: only expose trackEvent */
declare global {
  interface Window {
    trackEvent?: (name: string, data?: Record<string, unknown>) => void;
  }
}

/**
 * Ensure we only output one JSON-LD per @type/@id.
 * If no @id is present, we keep the first occurrence of that @type.
 */
function uniqueByTypeId(nodes: any[]): any[] {
  const seen = new Set<string>();
  return nodes.filter((n) => {
    if (!n || typeof n !== "object") return false;
    const rawType = Array.isArray(n["@type"]) ? n["@type"].join("|") : String(n["@type"] ?? "");
    const id = typeof n["@id"] === "string" ? n["@id"] : "";
    const key = `${rawType}#${id}`;
    if (seen.has(key)) return false;
    seen.add(key);
    return true;
  });
}

export async function getStaticPaths() {
  const posts: BlogEntry[] = await getCollection("blog");
  const published = toListing(posts); // removes drafts unless LOCAL_DRAFTS=1
  const tags = new Set<string>();
  for (const p of published) {
    const t = (p.data as any).tags;
    if (Array.isArray(t)) for (const x of t) tags.add(String(x));
  }
  return Array.from(tags).map((t) => ({ params: { tag: t } }));
}

export async function getStaticPaths() {
  const posts: BlogEntry[] = await getCollection("blog");
  const published = toListing(posts);
  const cats = new Set<string>();
  for (const p of published) {
    const c = (p.data as any).category;
    if (typeof c === "string" && c) cats.add(c);
  }
  return Array.from(cats).map((c) => ({ params: { category: c } }));
}
